664: /// INCLUDE size checks on the measured serial callback, with its actual unmodelled
665: /// heap state asserted. Error sizes are the PG18.6 a163_review_sizes fixture's.
666: #[test]
667: fn measured_serial_include_size() {
668:     let dir = tempfile::tempdir().unwrap();
669:     let eng = open(dir.path());
670:     with_max_versions(500, || {
671:         let mut s = eng.connect(1, "postgres", "postgres", "t").unwrap();
672:         s.query("SET max_parallel_maintenance_workers=0").unwrap();
673:         s.query("CREATE FUNCTION mi_text(n int) RETURNS text LANGUAGE sql IMMUTABLE AS $$ SELECT left(string_agg(md5(g::text), ''), n) FROM generate_series(1, (n+31)/32) g $$").unwrap();
674:         for (len, expected) in [(2800, "index row size 2816 exceeds btree version 4 maximum 2704"), (8300, "index row requires 8320 bytes, maximum size is 8191")] {
675:             s.query("CREATE TABLE mi (k int, payload text)").unwrap();
676:             s.query(&format!("INSERT INTO mi SELECT g, CASE WHEN g=6000 THEN mi_text({len}) ELSE 'small' END FROM generate_series(1,6000) g")).unwrap();
677:             s.query("VACUUM mi").unwrap();
678:             let result = s.query("SELECT oid FROM pg_class WHERE relname='mi'").unwrap();
679:             let pg_engine::types::Value::Oid(oid) = result[0].rows[0][0] else { panic!("table oid") };
680:             assert!(pg_engine::exec::sizing::is_estimated(oid, &s.state.db));
681:             assert!(pg_engine::exec::sizing::heap_writes(oid, &s.state.db).is_none());
682:             let e = s.query("CREATE INDEX mi_i ON mi(k) INCLUDE(payload)").unwrap_err();
683:             assert_eq!(e.code, "54000");
684:             assert!(e.message.starts_with(expected), "{e:?}");
685:             eprintln!("measured serial INCLUDE: estimated=true heap_model=false workers=0: {}", e.message);
686:             s.query("DROP TABLE mi").unwrap();
687:         }
688:     });
