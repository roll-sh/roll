# The NOT iterators' own timeout checks (`TimeoutContextClock`, every 5000 documents they
# skip): a cursor keeps its iterators, and with them the deadline they were built with, while
# every read restarts the request's clock. A read after that deadline passed times out in
# the NOT iterator once it skipped 5000 documents (ON_TIMEOUT RETURN: an empty chunk with the
# warning, the cursor kept); the plain NOT is then at EOF for good, the optimized one (INDEXALL)
# reads on from where it stopped. Under ON_TIMEOUT FAIL without workers the read fails and the
# cursor is gone. Indexes are created on an empty keyspace; configs are put back.
!delta search_coord_*,search_shard_*,search_total_queries_processed,search_total_query_commands INFO search_queries search_warnings_and_errors search_coordinator_warnings_and_errors
FLUSHALL
FT.CREATE nt ON HASH PREFIX 1 n: SCHEMA t TEXT
FT.CREATE na ON HASH PREFIX 1 n: INDEXALL ENABLE SCHEMA t TEXT
EVAL "redis.call('HSET','n:00000','t','x') for i=1,5100 do redis.call('HSET',string.format('n:%05d',i),'t','hello') end redis.call('HSET','n:99999','t','y') redis.call('HSET','n:99998','t','hello') redis.call('HSET','n:99997','t','z') return 1" 0
FT.SEARCH nt -hello NOCONTENT
FT.SEARCH na -hello NOCONTENT
!delta search_coord_*,search_shard_*,search_total_queries_processed,search_total_query_commands INFO search_queries search_warnings_and_errors search_coordinator_warnings_and_errors

# ON_TIMEOUT RETURN: the plain NOT ends.
