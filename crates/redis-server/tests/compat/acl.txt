# EXEC itself denied: the transaction is discarded
MULTI
GET cached:1
EXEC
@conn 1
ACL SETUSER alice +multi +exec
@conn 2

# denied while queueing: the transaction is aborted
MULTI
GET cached:1
GET other
PING
EXEC

# permissions changed after queueing: checked again at EXEC
MULTI
GET cached:1
SET wo:9 v
@conn 1
ACL SETUSER alice -set
@conn 2
EXEC

# rule changes apply to connected clients right away

ACL SETUSER alice -get

@conn 1
ACL SETUSER alice +get
@conn 2
