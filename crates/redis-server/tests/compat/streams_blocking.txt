# Streams: blocking XREAD / XREADGROUP across connections.

# ---- XREAD BLOCK with $ ----
DEL s s2 str
@conn 2
!send XREAD BLOCK 0 STREAMS s $
@conn 1
@sleep 50
XADD s 1-0 a 1
@conn 2
@recv
@conn 1
XADD s 2-0 a 2
@conn 2
!send XREAD BLOCK 0 STREAMS s $
@conn 1
@sleep 50
XADD s2 1-0 b 1
XADD s 3-0 a 3
@conn 2
@recv

# ---- Timeouts ----
XREAD BLOCK 50 STREAMS s $
XREAD BLOCK 50 STREAMS missing $
XREAD BLOCK 50 STREAMS s 3-0
XREAD BLOCK 50 STREAMS s 2-0
XREAD BLOCK 50 STREAMS s +
XREAD BLOCK 50 COUNT 1 STREAMS s 0
XREAD BLOCK 50 STREAMS s s2 $ $
@resp 3
XREAD BLOCK 50 STREAMS s $
XREAD BLOCK 50 STREAMS s 0
@resp 2
@conn 1

# ---- XREADGROUP BLOCK woken by XADD ----
DEL s
XGROUP CREATE s g $ MKSTREAM
@conn 2
!send XREADGROUP GROUP g c1 BLOCK 0 STREAMS s >
@sleep 50
@conn 3
!send XREADGROUP GROUP g c2 BLOCK 0 STREAMS s >
@conn 1
@sleep 50
XADD s 1-0 a 1
@conn 2
@recv
@conn 1
XADD s 2-0 a 2
@conn 3
@recv
@conn 1
XPENDING s g
XINFO GROUPS s
XREADGROUP GROUP g c1 BLOCK 50 STREAMS s >
XREADGROUP GROUP g c1 BLOCK 50 STREAMS s 0
XREADGROUP GROUP g c9 BLOCK 50 STREAMS s 0
@resp 3
XREADGROUP GROUP g c1 BLOCK 50 STREAMS s >
