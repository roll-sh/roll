# FT.SEARCH: INDEXALL, suffix tries, weights, expansion, expiration and deletion.

FLUSHALL
FT.CREATE a ON HASH PREFIX 1 a: INDEXALL ENABLE SCHEMA t TEXT WEIGHT 2 WITHSUFFIXTRIE u TEXT tg TAG WITHSUFFIXTRIE
HSET a:1 t "running runner" u "jumped" tg "alpha,beta"
HSET a:2 t "runs quickly" u "running" tg "gamma"
HSET a:3 x "unindexed"
HSET a:4 t "dogs barking" tg "alphabet"
FT.SEARCH a * NOCONTENT
FT.SEARCH a "-running" NOCONTENT
FT.SEARCH a "~running" NOCONTENT WITHSCORES SCORER TFIDF
FT.SEARCH a running NOCONTENT WITHSCORES SCORER TFIDF EXPLAINSCORE
FT.SEARCH a running VERBATIM NOCONTENT
FT.SEARCH a "@t:running" WITHSCORES SCORER BM25 NOCONTENT
FT.SEARCH a "@u:running" WITHSCORES SCORER BM25 NOCONTENT
FT.SEARCH a "(running) => {$weight: 3}" WITHSCORES SCORER TFIDF NOCONTENT DIALECT 2
FT.SEARCH a "*unn*" NOCONTENT
FT.SEARCH a "*ing" NOCONTENT
FT.SEARCH a "@u:*ing" NOCONTENT
FT.SEARCH a "w'*ark*'" NOCONTENT DIALECT 2
FT.SEARCH a "@tg:{*pha*}" NOCONTENT DIALECT 2
FT.SEARCH a "@tg:{*ta}" NOCONTENT DIALECT 2
FT.SEARCH a "@tg:{alpha*}" NOCONTENT
FT.SEARCH a "@tg:{w'al*a'}" NOCONTENT DIALECT 2
FT.SEARCH a dogs NOSTOPWORDS NOCONTENT
FT.SEARCH a "dog" LANGUAGE english NOCONTENT
FT.SEARCH a "the dogs" NOCONTENT
HDEL a:4 t
FT.SEARCH a * NOCONTENT
DEL a:2
FT.SEARCH a running NOCONTENT
FT.SEARCH a "-running" NOCONTENT
FT.DROPINDEX a
FT.CREATE e ON HASH PREFIX 1 e: SCHEMA t TEXT n NUMERIC tg TAG
HSET e:1 t "keep me" n 1 tg x
HSET e:2 t "keep me too" n 2 tg x
HPEXPIRE e:2 100000 FIELDS 1 t
FT.SEARCH e keep NOCONTENT
HPEXPIRE e:2 1 FIELDS 1 n
@sleep 50
FT.SEARCH e "@n:[0 10]" NOCONTENT
FT.SEARCH e "@tg:{x}" NOCONTENT
FT.SEARCH e keep NOCONTENT
FT.DROPINDEX e
FLUSHALL
