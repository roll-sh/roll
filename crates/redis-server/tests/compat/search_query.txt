# FT.SEARCH execution.

FLUSHALL
FT.CREATE idx ON HASH PREFIX 1 d: SCORE_FIELD sc PAYLOAD_FIELD pl SCHEMA t TEXT SORTABLE t2 TEXT n NUMERIC SORTABLE n2 NUMERIC tg TAG SORTABLE ct TAG CASESENSITIVE g GEO m TEXT INDEXMISSING
HSET d:1 t "hello world" t2 "foo bar" n 5 n2 1 tg "a,b" ct "Ab" g "2.35,48.85" sc 0.5 pl abc m x
HSET d:2 t "hello there world" t2 "bar baz" n 10 n2 2 tg "b,c" ct "ab" g "2.36,48.86" pl abd
HSET d:3 t "world peace" n 3 tg "c" ct "AB" g "13.4,52.5" m y
HSET d:4 t "hello hello hello" t2 "foo" n 7 n2 9 tg "a" pl xyz
HSET d:5 t "the quick brown fox jumps" t2 "lazy dog" n 1 tg "d,e" g "-0.12,51.5"
FT.SEARCH idx hello
FT.SEARCH idx hello NOCONTENT
FT.SEARCH idx hello WITHSCORES
FT.SEARCH idx hello WITHSCORES SCORER TFIDF
FT.SEARCH idx hello WITHSCORES SCORER TFIDF.DOCNORM
FT.SEARCH idx hello WITHSCORES SCORER BM25
FT.SEARCH idx hello WITHSCORES SCORER BM25STD
FT.SEARCH idx hello WITHSCORES SCORER BM25STD.TANH
FT.SEARCH idx hello WITHSCORES SCORER BM25STD.NORM
FT.SEARCH idx hello WITHSCORES SCORER DISMAX
FT.SEARCH idx hello WITHSCORES SCORER DOCSCORE
FT.SEARCH idx hello WITHSCORES SCORER HAMMING PAYLOAD abc
FT.SEARCH idx "hello world" WITHSCORES SCORER TFIDF EXPLAINSCORE
FT.SEARCH idx "hello|peace" WITHSCORES SCORER BM25 EXPLAINSCORE
FT.SEARCH idx "hello world" WITHSCORES EXPLAINSCORE
FT.SEARCH idx "hello world" WITHSCORES SCORER DISMAX EXPLAINSCORE
FT.SEARCH idx hello EXPLAINSCORE
FT.SEARCH idx hello WITHPAYLOADS
FT.SEARCH idx hello WITHSORTKEYS SORTBY n
FT.SEARCH idx hello WITHSORTKEYS SORTBY t DESC
FT.SEARCH idx * SORTBY n ASC
FT.SEARCH idx * SORTBY n DESC LIMIT 1 2
FT.SEARCH idx * SORTBY n2
FT.SEARCH idx * SORTBY t2
FT.SEARCH idx * SORTBY nosuch
FT.SEARCH idx * LIMIT 0 0
FT.SEARCH idx * LIMIT 0 2
FT.SEARCH idx * LIMIT 3 10
FT.SEARCH idx * NOCONTENT
FT.SEARCH idx * RETURN 2 t n
FT.SEARCH idx * RETURN 1 nosuch
FT.SEARCH idx * RETURN 3 t AS title
FT.SEARCH idx "hello -world"
FT.SEARCH idx "-hello" NOCONTENT
FT.SEARCH idx "~hello world" NOCONTENT WITHSCORES
FT.SEARCH idx "hel*" NOCONTENT
FT.SEARCH idx "*orld" NOCONTENT
FT.SEARCH idx "*orl*" NOCONTENT
FT.SEARCH idx "%helo%" NOCONTENT
FT.SEARCH idx "%%wrld%%" NOCONTENT
FT.SEARCH idx "w'*l*'" NOCONTENT DIALECT 2
FT.SEARCH idx "\"hello world\"" NOCONTENT
FT.SEARCH idx "\"world hello\"" NOCONTENT
FT.SEARCH idx "hello world" SLOP 0 NOCONTENT
FT.SEARCH idx "hello world" SLOP 1 INORDER NOCONTENT
FT.SEARCH idx "world hello" INORDER NOCONTENT
FT.SEARCH idx "@t:hello" NOCONTENT
FT.SEARCH idx "@t2:foo" NOCONTENT
FT.SEARCH idx "@t|t2:bar" NOCONTENT
FT.SEARCH idx bar INFIELDS 1 t2 NOCONTENT
FT.SEARCH idx bar INFIELDS 1 t NOCONTENT
FT.SEARCH idx hello INKEYS 2 d:1 d:4 NOCONTENT
FT.SEARCH idx "@tg:{a}" NOCONTENT
FT.SEARCH idx "@tg:{a|c}" NOCONTENT
FT.SEARCH idx "@tg:{A}" NOCONTENT
FT.SEARCH idx "@ct:{Ab}" NOCONTENT
FT.SEARCH idx "@ct:{ab}" NOCONTENT
FT.SEARCH idx "@ct:{A*}" NOCONTENT
FT.SEARCH idx "@tg:{*}" NOCONTENT
FT.SEARCH idx "@ct:{*b}" NOCONTENT DIALECT 2
FT.SEARCH idx "@ct:{*B*}" NOCONTENT DIALECT 2
FT.SEARCH idx "@ct:{w'?b'}" NOCONTENT DIALECT 2
FT.SEARCH idx "@n:[3 7]" NOCONTENT
FT.SEARCH idx "@n:[(3 (7]" NOCONTENT
FT.SEARCH idx "@n:[-inf 4]" NOCONTENT
FT.SEARCH idx "@n2:[1 2] @n:[1 100]" NOCONTENT
FT.SEARCH idx * FILTER n 5 10 NOCONTENT
FT.SEARCH idx "@g:[2.35 48.85 5 km]" NOCONTENT
FT.SEARCH idx "@g:[2.35 48.85 500 m]" NOCONTENT
FT.SEARCH idx "@g:[2.35 48.85 1000 km]" NOCONTENT
FT.SEARCH idx "@g:[2.35 48.85 3000 mi]" NOCONTENT
FT.SEARCH idx * GEOFILTER g 2.35 48.85 2 km NOCONTENT
FT.SEARCH idx "ismissing(@m)" NOCONTENT DIALECT 2
FT.SEARCH idx "-ismissing(@m)" NOCONTENT DIALECT 2
FT.SEARCH idx "hello ~foo" WITHSCORES SCORER TFIDF NOCONTENT
FT.SEARCH idx "(hello|world) -peace @n:[1 6]" NOCONTENT
FT.SEARCH idx "@t:(hello world) | @t2:baz" NOCONTENT
FT.SEARCH idx "hello" SORTBY n LIMIT 0 2 WITHSCORES
FT.SEARCH idx nosuchterm
FT.SEARCH nosuch hello
FT.SEARCH idx "@t:hel*" WITHSCORES SCORER TFIDF EXPLAINSCORE NOCONTENT
FT.SEARCH idx "@t:[a c]" NOCONTENT
FT.SEARCH idx "@n:[0 +inf]" WITHSCORES SCORER BM25STD NOCONTENT
@resp 3
FT.SEARCH idx hello WITHSCORES WITHPAYLOADS RETURN 1 t
FT.SEARCH idx hello WITHSORTKEYS SORTBY n NOCONTENT
FT.SEARCH idx * LIMIT 0 0
FT.SEARCH idx "hel*" LIMIT 0 1
@resp 2
FT.DROPINDEX idx
FLUSHALL
