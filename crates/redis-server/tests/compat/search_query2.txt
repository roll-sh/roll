# FT.SEARCH over JSON, databases, big unions and expansion limits.

FLUSHALL
FT.CREATE j ON JSON PREFIX 1 j: SCHEMA $.name AS name TEXT $.n AS n NUMERIC SORTABLE $.tags[*] AS tags TAG $.loc AS loc GEO
JSON.SET j:1 $ "{\"name\":\"alpha beta\",\"n\":3,\"tags\":[\"x\",\"y\"],\"loc\":\"2.35,48.85\",\"o\":{\"a\":1}}"
JSON.SET j:2 $ "{\"name\":\"beta gamma\",\"n\":1.5,\"tags\":[\"y\"],\"o\":[1,2]}"
FT.SEARCH j beta
FT.SEARCH j beta DIALECT 3
FT.SEARCH j beta RETURN 1 name
FT.SEARCH j beta RETURN 1 $.o
FT.SEARCH j beta RETURN 1 $.o DIALECT 3
FT.SEARCH j beta RETURN 3 $.tags AS t n
FT.SEARCH j beta RETURN 2 $.tags[*] __key DIALECT 3
FT.SEARCH j "@tags:{y}" SORTBY n NOCONTENT
FT.SEARCH j * SORTBY n DESC RETURN 1 n
FT.SEARCH j "@loc:[2.35 48.85 10 km]" NOCONTENT
@resp 3
FT.SEARCH j beta RETURN 1 $.tags[*] DIALECT 3
FT.SEARCH j beta RETURN 1 $.o FORMAT EXPAND DIALECT 3
@resp 2
FT.DROPINDEX j
FT.CREATE w ON HASH PREFIX 1 w: SCHEMA t TEXT
HSET w:1 t "w01 w02 w03 w04 w05 w06 w07 w08 w09 w10 w11 w12"
HSET w:2 t "w13 w14 w15 w16 w17 w18 w19 w20 w21 w22 w23 w24 w25"
HSET w:3 t "w05 w13 w25 other"
HSET w:4 t "nothing here"
FT.SEARCH w "w*" NOCONTENT WITHSCORES SCORER TFIDF
FT.SEARCH w "w*" NOCONTENT WITHSCORES SCORER DISMAX
FT.SEARCH w "w* other" NOCONTENT WITHSCORES SCORER BM25
FT.SEARCH w "-w*" NOCONTENT
FT.SEARCH w "w01|w13|w25|w05|other" WITHSCORES SCORER TFIDF NOCONTENT
FT.SEARCH w "w01 w02 w03" WITHSCORES SCORER TFIDF EXPLAINSCORE NOCONTENT
FT.SEARCH w "w05 w13 w25" WITHSCORES SCORER BM25 EXPLAINSCORE NOCONTENT
FT.SEARCH w "w01 w03" SLOP 0 NOCONTENT
FT.SEARCH w "w01 w03" SLOP 1 NOCONTENT
FT.SEARCH w "w03 w01" SLOP 1 INORDER NOCONTENT
FT.SEARCH w "~w05 w13" NOCONTENT WITHSCORES SCORER TFIDF
FT.CONFIG SET MAXPREFIXEXPANSIONS 5
FT.SEARCH w "w*" NOCONTENT
FT.SEARCH w "w1*" NOCONTENT
@resp 3
FT.SEARCH w "w*" NOCONTENT
FT.SEARCH w "%w0%" NOCONTENT
@resp 2
FT.CONFIG SET MAXPREFIXEXPANSIONS 200
FT.CONFIG SET UNION_ITERATOR_HEAP 2
FT.SEARCH w "w05|w13|w25" WITHSCORES SCORER TFIDF NOCONTENT
FT.SEARCH w "w*" WITHSCORES SCORER BM25 NOCONTENT
FT.CONFIG SET UNION_ITERATOR_HEAP 20
FT.DROPINDEX w
FT.CREATE s ON HASH PREFIX 1 s: SCHEMA t TEXT
SELECT 1
HSET s:1 t "in db one"
FT.SEARCH s one
SELECT 0
HSET s:2 t "in db zero"
FT.SEARCH s "in"
SELECT 1
FT.SEARCH s "in"
SELECT 0
FT.DROPINDEX s
SELECT 1
FLUSHDB
SELECT 0
