# A query or a deletion creates the index.
FLUSHALL
FT.CREATE gq ON HASH PREFIX 1 gq: SCHEMA g GEOSHAPE t TEXT
FT.SEARCH gq "@g:[WITHIN $s]" PARAMS 2 s "POINT(1 x)" DIALECT 3
!mask total_indexing_time FT.INFO gq
FLUSHALL
FT.CREATE gq2 ON HASH PREFIX 1 gq2: SCHEMA g GEOSHAPE t TEXT
HSET gq2:1 t hello
!mask total_indexing_time FT.INFO gq2
DEL gq2:1
!mask total_indexing_time FT.INFO gq2
FT.DROPINDEX gq2
FLUSHALL
