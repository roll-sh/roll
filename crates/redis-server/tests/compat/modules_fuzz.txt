# ---- A TopK heap slot holding the empty item loads without one (NULL): TOPK.LIST answers
# a null for it, and the empty item still matches it (memcmp of 0 bytes) ----
TOPK.RESERVE k 3 50 3 1
TOPK.ADD k "a b" i110 "" i60 "" i48 "" b
TOPK.LIST k WITHCOUNT
EVAL "local d = redis.call('DUMP', KEYS[1]); return redis.call('RESTORE', KEYS[2], 0, d)" 2 k k2
TOPK.LIST k2 WITHCOUNT
TOPK.LIST k2
TOPK.QUERY k2 "" i48 x
TOPK.COUNT k2 "" i48
TOPK.ADD k2 "" ""
TOPK.LIST k2 WITHCOUNT
@resp 3
TOPK.LIST k2 WITHCOUNT
@resp 2
FLUSHALL
