# FT.AGGREGATE GROUPBY and the reducers. Groups come out in the module's hash order, so
# every grouped query sorts on a unique key (or has a single group).

FLUSHALL
FT.CREATE idx ON HASH PREFIX 1 d: SCHEMA name TEXT SORTABLE age NUMERIC SORTABLE city TAG tags TAG SORTABLE score NUMERIC body TEXT nick TEXT SORTABLE UNF
HSET d:1 name alice age 30 city NYC tags "a,b" score 1.5 body "hello world" nick Al
HSET d:2 name bob age 25 city LA tags b score 2 body "hello there" nick Bo
HSET d:3 name carol age 35 city NYC score 3.25 extra x body "world peace"
HSET d:4 name dave age 40 city SF tags "c,a" score -1 body "hello hello" nick Da
HSET d:5 name eve age 22 city LA tags "" score 10 body goodbye
HSET d:6 name 5 age 28 city NYC tags "b,c" extra y
FT.AGGREGATE idx * GROUPBY 1 @city REDUCE COUNT 0 AS n SORTBY 2 @city ASC
