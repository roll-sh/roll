# ---- an empty index of every field type ----
FT.CREATE every ON HASH PREFIX 2 e: f: FILTER "@n>1" LANGUAGE german LANGUAGE_FIELD lang SCORE 0.5 SCORE_FIELD sc PAYLOAD_FIELD pl NOOFFSETS NOFREQS MAXTEXTFIELDS STOPWORDS 2 foo bar INDEXALL ENABLE SCHEMA t AS tt TEXT WEIGHT 2 NOSTEM PHONETIC dm:en WITHSUFFIXTRIE INDEXEMPTY INDEXMISSING SORTABLE UNF n NUMERIC SORTABLE NOINDEX tg TAG SEPARATOR "|" WITHSUFFIXTRIE g GEO INDEXMISSING v VECTOR HNSW 6 TYPE FLOAT32 DIM 4 DISTANCE_METRIC COSINE fl VECTOR FLAT 6 TYPE FLOAT64 DIM 2 DISTANCE_METRIC IP s VECTOR SVS-VAMANA 6 TYPE FLOAT32 DIM 4 DISTANCE_METRIC L2 gs GEOSHAPE FLAT
FT.CREATE every SCHEMA t TEXT
FT._CREATEIFNX every SCHEMA t TEXT
!mask total_indexing_time FT.INFO every
FT.INFO every
FT.CREATE plain SCHEMA t TEXT NOSTEM
!sort FT._LIST
!mask total_indexing_time FT.INFO plain
FT.CREATE nofields NOFIELDS NOHL SCORE 0 LANGUAGE english SCHEMA t TEXT tg TAG CASESENSITIVE SORTABLE

FT.DROPINDEX every

FT.DROPINDEX nofields
!sort FT._LIST

# ---- indexing hashes ----
FT.CREATE idx ON HASH PREFIX 1 doc: SCHEMA t TEXT SORTABLE n NUMERIC tg TAG SEPARATOR ; CASESENSITIVE g GEO
HSET doc:1 t "hello world running" n 5 tg "a;B" g "1.5,2.5"
HSET doc:2 t "foo bar" n 7
HSET other:1 t "not indexed"
!mask total_indexing_time FT.INFO idx
# Failures: the document is dropped and the error recorded on the index and the field.
HSET doc:3 t "x" n abc
HSET doc:4 g "500,10"
HSET doc:5 g "abc"
HSET doc:6 g "12"
!mask total_indexing_time FT.INFO idx
# An update replaces the document (new id).
HSET doc:1 t "hello again"
HSET doc:2 n 8 tg "x;y;x"
!mask total_indexing_time FT.INFO idx
# Fields the schema doesn't know don't reindex.
HSET doc:2 other v
HINCRBY doc:2 n 1
HINCRBYFLOAT doc:2 n 0.5
HSETNX doc:2 t "ignored"
HSETNX doc:2 newfield v
HDEL doc:2 nosuch
HDEL doc:2 tg
!mask total_indexing_time FT.INFO idx
# Deletes, renames, TTLs.
DEL doc:1
RENAME doc:2 doc:22
RENAME doc:22 other:22
RENAME other:22 doc:2
EXPIRE doc:2 1000
PERSIST doc:2
COPY doc:2 doc:7
COPY doc:2 other:8
SET doc:7 overwritten
!mask total_indexing_time FT.INFO idx
# Events from other databases index the key from there.
SELECT 1
HSET doc:9 t "in db one"
DEL doc:9
HSET doc:10 t "stays"
MOVE doc:10 0
SELECT 0
!mask total_indexing_time FT.INFO idx
HDEL doc:2 t n
DEL doc:2
!mask total_indexing_time FT.INFO idx

# ---- FILTER rules: the fields are read by name (or path), unknown properties by their own name ----
FLUSHALL
FT.CREATE fi ON HASH PREFIX 1 f: FILTER "@n>1 && @extra!='no'" SCHEMA num AS n NUMERIC t TEXT
HSET f:1 num 0 t a
HSET f:2 num 5 t b
HSET f:3 num 5 t c extra no
HSET f:4 t d
!mask total_indexing_time FT.INFO fi
HSET f:2 num 0
HSET f:3 extra yes
!mask total_indexing_time FT.INFO fi
FT.CREATE fk ON HASH FILTER "startswith(@__key, 'k:a')" SCHEMA t TEXT
HSET k:a1 t x
HSET k:b1 t y
HSET k:a2 __key k:b t z
!mask total_indexing_time FT.INFO fk
RENAME k:a1 k:b2
RENAME k:b1 k:a3
!mask total_indexing_time FT.INFO fk
FT.CREATE fe ON HASH PREFIX 1 e: FILTER "exists(@t)" SCHEMA t TEXT
HSET e:1 t a
HSET e:2 u a
!mask total_indexing_time FT.INFO fe
