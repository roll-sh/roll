# LOLWUT: version 8's TAPE MARK I by default, 5's Schotter and 6's skyline by VERSION, and the
# version line for any other version. The art is random, so whole replies are compared by shape
# and the deterministic parts exactly (Schotter's first two rows of squares aren't jittered).

!type LOLWUT
!type LOLWUT IT
!type LOLWUT it
!type LOLWUT VERSION
!type LOLWUT VERSION 8
!type LOLWUT VERSION 8 IT
!type LOLWUT VERSION 5
!type LOLWUT VERSION 6
!type LOLWUT VERSION 6 10 5
EVAL "return string.sub(redis.call('lolwut'), -40)" 0
EVAL "return string.sub(redis.call('lolwut', 'IT'), -40)" 0
EVAL "local s = redis.call('lolwut') return string.find(s, '/', 1, true) == nil and s == string.upper(s) and 1 or 0" 0
EVAL "local s = redis.call('lolwut', 'version', '6', '7', '3') return string.sub(s, -110)" 0
EVAL "local s = redis.call('lolwut', 'version', '6', '7', '3') local n = 0 for _ in string.gmatch(s, '\27%[0m') do n = n + 1 end return n" 0
EVAL "local s = redis.call('lolwut', 'version', '6', '0', '0') local n = 0 for _ in string.gmatch(s, '\27%[0m') do n = n + 1 end return n" 0
