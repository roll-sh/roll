# ---- LATENCY HISTOGRAM ----
CONFIG RESETSTAT
!mask histogram_usec LATENCY HISTOGRAM
LATENCY HISTOGRAM nosuch
LATENCY HISTOGRAM nosuch|sub
LATENCY HISTOGRAM a|b|c
SET k v
GET k
GET k
INCR n
CONFIG GET maxmemory
CONFIG GET maxmemory
!mask histogram_usec LATENCY HISTOGRAM set
!mask histogram_usec LATENCY HISTOGRAM GET SET
!mask histogram_usec LATENCY HISTOGRAM get get
!mask histogram_usec LATENCY HISTOGRAM incr nosuch get
!mask histogram_usec LATENCY HISTOGRAM config|get
!mask histogram_usec LATENCY HISTOGRAM CONFIG|GET
!mask histogram_usec LATENCY HISTOGRAM config|nosuch
!mask histogram_usec LATENCY HISTOGRAM get|sub
!mask histogram_usec LATENCY HISTOGRAM latency
# Module commands answer under their declared names; a container lists its subcommands.
FT.CREATE lhidx SCHEMA t TEXT
FT.SEARCH lhidx foo
VADD lhvs VALUES 2 1 2 a
!mask histogram_usec LATENCY HISTOGRAM ft.search FT.CREATE vadd
OBJECT ENCODING k
!mask histogram_usec LATENCY HISTOGRAM object
FT.DROPINDEX lhidx
DEL lhvs
@resp 3
!mask histogram_usec LATENCY HISTOGRAM set get
@resp 2
CONFIG SET latency-tracking no
GET k
!mask histogram_usec LATENCY HISTOGRAM get
CONFIG SET latency-tracking yes
GET k
!mask histogram_usec LATENCY HISTOGRAM get
CONFIG RESETSTAT
LATENCY HISTOGRAM get
