# FT.PROFILE SEARCH|AGGREGATE [LIMITED] QUERY: the command's own reply, then the single shard's
# profile: the iterator tree (types, terms, ranges, reading operations, estimates, union query
# types, LIMITED's collapsed unions) and the result processors with the results each passed.
# Wall-clock fields are masked (`!timing`).

FLUSHALL
FT.CREATE idx ON HASH PREFIX 1 d SCHEMA t TEXT n NUMERIC SORTABLE g TAG loc GEO opt TAG INDEXMISSING
FT.CREATE ia ON HASH PREFIX 1 d INDEXALL ENABLE SCHEMA t TEXT n NUMERIC
HSET d1 t "hello world" n 1 g a loc "1,1" opt x
HSET d2 t "hello there" n 2 g b loc "1.5,1.5"
HSET d3 t "help me" n 3 g a,c loc "10,10"
HSET d4 t "world peace" n 4 g c loc "1.01,1.01" opt y
HSET d5 t "goodbye world" n 5 g ab loc "2,2"
HSET d6 t "hello again" n 6 g b
HSET d7 t "helium balloon" n 7 g abc
HSET d8 t "hello hello hello" n 8 g a

# The arguments.
FT.PROFILE
FT.PROFILE idx SEARCH QUERY
FT.PROFILE idx SEARCH QUERY x withcursor
FT.PROFILE idx AGGREGATE QUERY * WITHCURSOR COUNT 1
FT.PROFILE nosuch SEARCH QUERY x
FT.PROFILE nosuch SEARCH QUERY x WITHCURSOR
FT.PROFILE idx FOO QUERY x
FT.PROFILE idx WITHCURSOR QUERY x
FT.PROFILE idx search limited x
FT.PROFILE idx search LIMITED QUERY
FT.PROFILE idx SEARCH LIMITED LIMITED x
FT.PROFILE idx SEARCH x QUERY
FT.PROFILE idx aggregate query * y
FT.PROFILE idx search query hello y
FT.PROFILE idx search querY hello limit 0
FT.PROFILE idx SEARCH QUERY "hello(" NOCONTENT
FT.PROFILE idx SEARCH QUERY hello SORTBY nosuch
